[BEGIN]
module TopModule(
  input in,
  input [3:0] state,
  output [3:0] next_state,
  output out
);
  // One-hot encoded 4-state Moore FSM
  localparam A = 0, B = 1, C = 2, D = 3;

  // State transitions (Moore machine, one-hot):
  //   A --in=0--> A,  A --in=1--> B
  //   B --in=0--> C,  B --in=1--> B
  //   C --in=0--> A,  C --in=1--> D
  //   D --in=0--> C,  D --in=1--> B
  assign next_state[A] = ~in & (state[A] | state[C]);
  assign next_state[B] =  in & (state[A] | state[B] | state[D]);
  assign next_state[C] = ~in & (state[B] | state[D]);
  assign next_state[D] =  in &  state[C];

  // Output is high only in state D
  assign out = state[D];
endmodule
[DONE]
