[BEGIN]
module TopModule(
  input clk,
  input [7:0] in,
  input reset,
  output done
);
  // 4-bit state register with synchronous reset.
  // Reset loads state = 4'b0001 (state[0] set, all other bits cleared).
  reg [3:0] state;

  // Condition used by the next-state logic: state[0] or state[3] is active
  wire active = state[0] | state[3];

  always @(posedge clk) begin
    if (reset) begin
      state <= 4'b0001;
    end else begin
      state[0] <= active & ~in[3];  // stays/returns when in[3] is low
      state[1] <= state[2];
      state[2] <= active & in[3];   // advances when in[3] is high
      state[3] <= state[1];
    end
  end

  // done is asserted when the top state bit is set
  assign done = state[3];
endmodule
[DONE]
