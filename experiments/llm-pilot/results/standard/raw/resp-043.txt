[BEGIN]
module TopModule(
  input clk,
  input in,
  output reg out
);
  // Toggle flip-flop: the next state is in XOR out.
  // When in is 1 the output toggles on each rising clock edge;
  // when in is 0 the output holds its value.
  always @(posedge clk) begin
    out <= in ^ out;
  end
endmodule
[DONE]
