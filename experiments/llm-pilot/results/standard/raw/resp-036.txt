[BEGIN]
module TopModule(
  input c,
  input d,
  output [3:0] mux_in
);
  // Karnaugh-map-to-4:1-mux style design: c and d are the K-map
  // column/row select inputs, and mux_in holds the data inputs of the
  // 4-to-1 multiplexer that implements the function.
  assign mux_in[0] = c | d;        // c=0, d=0 case
  assign mux_in[1] = 1'b0;         // c=0, d=1 case
  assign mux_in[2] = ~d;           // c=1, d=0 case
  assign mux_in[3] = c & d;        // c=1, d=1 case
endmodule
[DONE]
