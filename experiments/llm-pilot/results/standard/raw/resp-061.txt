[BEGIN]
module TopModule(
  input a,
  input b,
  input c,
  input d,
  output out_sop,
  output out_pos
);
  // Both outputs require c = 1 and the condition (b -> d), i.e. (~b | d).
  // The remaining term differs between the two outputs.
  wire c_and_b_implies_d = c & (~b | d);

  // out_sop: additionally (a -> d)
  assign out_sop = c_and_b_implies_d & (~a | d);

  // out_pos: additionally (a -> b)
  assign out_pos = c_and_b_implies_d & (b | ~a);
endmodule
[DONE]
