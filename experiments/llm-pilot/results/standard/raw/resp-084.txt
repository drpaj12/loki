[BEGIN]
module TopModule(
  input clk,
  input [7:0] in,
  output reg [7:0] pedge
);
  // Per-bit rising edge detector.
  // pedge[i] is 1 for one cycle after in[i] goes from 0 to 1.
  reg [7:0] in_prev;

  always @(posedge clk) begin
    in_prev <= in;
    pedge   <= in & ~in_prev;
  end
endmodule
[DONE]
