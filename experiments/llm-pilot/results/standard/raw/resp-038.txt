[BEGIN]
module TopModule(
  input a,
  input b,
  input c,
  input d,
  output out,
  output out_n
);
  // Two 2-input AND terms combined with OR (AND-OR structure).
  // The netlist builds this from NAND gates: NAND(NAND(a,b), NAND(c,d)) = (a&b) | (c&d)
  assign out = (a & b) | (c & d);

  // out_n is the inverted output
  assign out_n = ~out;
endmodule
[DONE]
