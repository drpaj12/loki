[BEGIN]
module TopModule(
  input a,
  input b,
  input c,
  input d,
  output out_sop,
  output out_pos
);
  // Two forms of a 4-input function that depends on c being high.
  // out_sop is 1 for minterms 2, 7, 15 (a b c d = 0010, 0111, 1111)
  // out_pos additionally accepts minterm 3 (0011), so the two outputs differ.

  // Sum of products: (c & d) | (~a & ~b & c)
  assign out_sop = (c & d) | (~a & ~b & c);

  // Product of sums: c & (~a | b) & (~b | d)
  assign out_pos = c & (~a | b) & (~b | d);
endmodule
[DONE]
