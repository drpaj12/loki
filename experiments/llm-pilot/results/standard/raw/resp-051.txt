[BEGIN]
module TopModule(
  input clk,
  input j,
  input k,
  input areset,
  output out
);
  // Two-state Moore machine with asynchronous reset to OFF.
  // OFF -> ON when j is high, ON -> OFF when k is high.
  // The output is high only in the ON state.
  localparam OFF = 1'b0;
  localparam ON  = 1'b1;

  reg state, next_state;

  // Next-state logic
  always @(*) begin
    case (state)
      OFF:     next_state = j ? ON : OFF;
      ON:      next_state = k ? OFF : ON;
      default: next_state = OFF;
    endcase
  end

  // State register with asynchronous reset
  always @(posedge clk or posedge areset) begin
    if (areset)
      state <= OFF;
    else
      state <= next_state;
  end

  // Moore output
  assign out = (state == ON);
endmodule
[DONE]
