[BEGIN]
module TopModule(
  input clk,
  input a,
  output reg [2:0] q
);
  // 3-bit counter with synchronous reset (a) to 4.
  // Counts 4 -> 5 -> 6 -> 0 -> 1 -> 2 -> 3 -> 4 ...
  // State 6 wraps to 0 instead of continuing to 7.
  always @(posedge clk) begin
    if (a)
      q <= 3'd4;
    else if (q == 3'd6)
      q <= 3'd0;
    else
      q <= q + 3'd1;
  end
endmodule
[DONE]
