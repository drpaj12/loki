[BEGIN]
module TopModule(
  input a,
  input b,
  input c,
  input d,
  output out_sop,
  output out_pos
);
  // Inputs are the bits of {a,b,c,d}.
  // Output is 1 for input values 2, 7, 15.
  // Output is 0 for input values 0, 1, 4, 5, 6, 9, 10, 13, 14.
  // Input values 3, 8, 11, 12 are don't-cares.

  // Sum of products
  assign out_sop = (c & d) | (~a & ~b & c);

  // Product of sums
  assign out_pos = c & (~a | b) & (~b | d);
endmodule
[DONE]
