[BEGIN]
module TopModule(
  input clk,
  input reset,
  output reg [31:0] q
);
  // 32-bit Galois LFSR with taps at bits 32, 22, 2, 1 (feedback from q[0]).
  // Synchronous reset loads 32'h1.
  wire [31:0] q_next;

  assign q_next = {q[0], q[31:1]} ^ ({32{q[0]}} & 32'h0020_0003);
  // Equivalent per-bit form:
  //   q_next[31]    = q[0]
  //   q_next[21]    = q[22] ^ q[0]
  //   q_next[1]     = q[2]  ^ q[0]
  //   q_next[0]     = q[1]  ^ q[0]
  //   other bits shift down: q_next[i] = q[i+1]

  always @(posedge clk) begin
    if (reset)
      q <= 32'h1;
    else
      q <= q_next;
  end
endmodule
[DONE]
