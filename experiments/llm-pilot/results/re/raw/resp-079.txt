[BEGIN]
module TopModule(
  input clk,
  input a,
  output reg [2:0] q
);
  // Synchronous counter: a loads 4, otherwise count 4,5,6 then wrap to 0
  always @(posedge clk) begin
    if (a)
      q <= 3'd4;
    else if (q == 3'd6)
      q <= 3'd0;
    else
      q <= q + 3'd1;
  end
endmodule
[DONE]
