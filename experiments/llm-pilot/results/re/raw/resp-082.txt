[BEGIN]
module TopModule(
  input clk,
  input [7:0] in,
  output reg [7:0] pedge
);
  // Rising-edge detector per bit: pedge is 1 for one cycle after
  // a bit goes from 0 to 1.
  reg [7:0] d_last;

  always @(posedge clk) begin
    d_last <= in;
    pedge  <= in & ~d_last;
  end
endmodule
[DONE]
