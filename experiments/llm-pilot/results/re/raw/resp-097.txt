[BEGIN]
module TopModule(
  input clk,
  input shift_ena,
  input count_ena,
  input data,
  output reg [3:0] q
);
  // 4-bit register with two modes:
  //  - shift_ena: shift left, data enters at the LSB (takes priority)
  //  - count_ena: decrement the value
  //  - otherwise: hold
  always @(posedge clk) begin
    if (shift_ena)
      q <= {q[2:0], data};
    else if (count_ena)
      q <= q - 4'd1;
  end
endmodule
[DONE]
