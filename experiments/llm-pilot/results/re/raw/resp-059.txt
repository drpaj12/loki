[BEGIN]
module TopModule(
  input in,
  input [1:0] state,
  output [1:0] next_state,
  output out
);
  // Moore output: asserted only in state 3
  assign out = (state == 2'd3);

  // Next-state logic
  assign next_state[0] = in;
  assign next_state[1] = (state[0] & ~in) | (~state[0] & state[1] & in);
endmodule
[DONE]
