[BEGIN]
module TopModule(
  input a,
  input b,
  input c,
  input d,
  output out_sop,
  output out_pos
);
  // Truth table over {a,b,c,d}:
  //   output 1 for 2, 7, 15
  //   output 0 for 0, 1, 4, 5, 6, 9, 10, 13, 14
  //   don't care for 3, 8, 11, 12

  // Sum of products
  assign out_sop = (c & d) | (~a & ~b & c);

  // Product of sums
  assign out_pos = c & (~a | b) & (~b | d);
endmodule
[DONE]
