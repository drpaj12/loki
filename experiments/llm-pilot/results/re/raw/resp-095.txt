[BEGIN]
module TopModule(
  input clk,
  input reset,
  input in,
  output disc,
  output flag,
  output err
);
  // HDLC bit-stream framing detector (synchronous reset).
  // Counts consecutive 1's:
  //   0 ones followed by 0 -> none
  //   5 ones then 0  -> disc (bit-stuffed 0 to discard)
  //   6 ones then 0  -> flag (01111110)
  //   7 or more ones -> err (until a 0 arrives)
  localparam NONE = 4'd0,
             ONE  = 4'd1,
             TWO  = 4'd2,
             THREE = 4'd3,
             FOUR = 4'd4,
             FIVE = 4'd5,
             SIX  = 4'd6,
             DISC = 4'd7,
             FLAG = 4'd8,
             ERR  = 4'd9;

  reg [3:0] state, next;

  always @(*) begin
    case (state)
      NONE:  next = in ? ONE   : NONE;
      ONE:   next = in ? TWO   : NONE;
      TWO:   next = in ? THREE : NONE;
      THREE: next = in ? FOUR  : NONE;
      FOUR:  next = in ? FIVE  : NONE;
      FIVE:  next = in ? SIX   : DISC;
      SIX:   next = in ? ERR   : FLAG;
      DISC:  next = in ? ONE   : NONE;
      FLAG:  next = in ? ONE   : NONE;
      ERR:   next = in ? ERR   : NONE;
      default: next = NONE;
    endcase
  end

  always @(posedge clk) begin
    if (reset)
      state <= NONE;
    else
      state <= next;
  end

  assign disc = (state == DISC);
  assign flag = (state == FLAG);
  assign err  = (state == ERR);
endmodule
[DONE]
