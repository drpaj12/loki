[BEGIN]
module TopModule(
  input clk,
  input reset,
  input ena,
  output reg pm,
  output reg [7:0] hh,
  output reg [7:0] mm,
  output reg [7:0] ss
);
  // 12-hour BCD clock with AM/PM flag, synchronous reset to 12:00:00 AM
  wire ss_wrap = (ss == 8'h59);
  wire mm_wrap = (mm == 8'h59);

  // BCD increment of an 8-bit two-digit value
  function [7:0] bcd_inc(input [7:0] v);
    begin
      if (v[3:0] == 4'd9)
        bcd_inc = {v[7:4] + 4'd1, 4'd0};
      else
        bcd_inc = {v[7:4], v[3:0] + 4'd1};
    end
  endfunction

  always @(posedge clk) begin
    if (reset) begin
      pm <= 1'b0;
      hh <= 8'h12;
      mm <= 8'h00;
      ss <= 8'h00;
    end else if (ena) begin
      // seconds
      if (ss_wrap) ss <= 8'h00;
      else         ss <= bcd_inc(ss);

      // minutes
      if (ss_wrap) begin
        if (mm_wrap) mm <= 8'h00;
        else         mm <= bcd_inc(mm);
      end

      // hours
      if (ss_wrap && mm_wrap) begin
        case (hh)
          8'h12:   hh <= 8'h01;
          8'h11:   begin
                     hh <= 8'h12;
                     pm <= ~pm;
                   end
          default: hh <= bcd_inc(hh);
        endcase
      end
    end
  end
endmodule
[DONE]
