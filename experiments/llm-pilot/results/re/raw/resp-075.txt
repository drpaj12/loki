[BEGIN]
module TopModule(
  input clk,
  input in,
  input reset,
  output done
);
  // Serial receiver FSM: start bit (0), 8 data bits, stop bit (1).
  // done is asserted after a valid stop bit is received.
  localparam IDLE  = 3'd0,
             DATA  = 3'd1,
             STOP  = 3'd2,
             DONE  = 3'd3,
             ERROR = 3'd4;

  reg [2:0] state;
  reg [3:0] count;

  always @(posedge clk) begin
    if (reset) begin
      state <= IDLE;
      count <= 4'd0;
    end else begin
      case (state)
        IDLE: begin
          if (!in) begin
            state <= DATA;
            count <= 4'd0;
          end
        end
        DATA: begin
          count <= count + 4'd1;
          if (count == 4'd7)
            state <= STOP;
        end
        STOP: begin
          if (in)
            state <= DONE;
          else
            state <= ERROR;
        end
        DONE: begin
          if (in)
            state <= IDLE;
          else begin
            state <= DATA;
            count <= 4'd0;
          end
        end
        ERROR: begin
          if (in)
            state <= IDLE;
        end
        default: state <= IDLE;
      endcase
    end
  end

  assign done = (state == DONE);
endmodule
[DONE]
