[BEGIN]
module TopModule(
  input clk,
  input j,
  input k,
  input areset,
  output out
);
  // Two-state Moore FSM with asynchronous reset to OFF
  // OFF -> ON when j is high, ON -> OFF when k is high
  localparam OFF = 1'b0, ON = 1'b1;

  reg state, next_state;

  always @(*) begin
    case (state)
      OFF:     next_state = j ? ON : OFF;
      ON:      next_state = k ? OFF : ON;
      default: next_state = OFF;
    endcase
  end

  always @(posedge clk or posedge areset) begin
    if (areset)
      state <= OFF;
    else
      state <= next_state;
  end

  assign out = (state == ON);
endmodule
[DONE]
