[BEGIN]
module TopModule(
  input clk,
  input d,
  input r,
  output reg q
);
  // D flip-flop with synchronous reset: q is cleared when r is high,
  // otherwise it captures d on the rising clock edge.
  always @(posedge clk) begin
    if (r)
      q <= 1'b0;
    else
      q <= d;
  end
endmodule
[DONE]
