[BEGIN]
module TopModule(
  input clk,
  input areset,
  input bump_left,
  input bump_right,
  output walk_left,
  output walk_right
);
  // Lemmings walking left/right FSM, one state bit.
  // state = 0: walking left, state = 1: walking right.
  // Asynchronous reset to walking left.
  reg state;

  always @(posedge clk or posedge areset) begin
    if (areset)
      state <= 1'b0;
    else if (state == 1'b0)
      state <= bump_left ? 1'b1 : 1'b0;   // turn right when bumped on the left
    else
      state <= bump_right ? 1'b0 : 1'b1;  // turn left when bumped on the right
  end

  assign walk_left  = ~state;
  assign walk_right = state;
endmodule
[DONE]
