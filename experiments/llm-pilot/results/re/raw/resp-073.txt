[BEGIN]
module TopModule(
  input clk,
  input [7:0] in,
  input reset,
  output done
);
  // One-hot FSM driven by in[3] only (the other input bits are unused).
  // state[0] is the idle/wait state entered on synchronous reset.
  reg [3:0] state;
  wire in3 = in[3];

  always @(posedge clk) begin
    if (reset) begin
      state <= 4'b0001;
    end else begin
      // state 0 or state 3: go to state 2 if in3, otherwise stay in state 0
      state[0] <= ~in3 & (state[0] | state[3]);
      state[1] <= state[2];
      state[2] <= in3 & (state[0] | state[3]);
      state[3] <= state[1];
    end
  end

  assign done = state[3];
endmodule
[DONE]
