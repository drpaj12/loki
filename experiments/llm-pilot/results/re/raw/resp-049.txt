[BEGIN]
module TopModule(
  input clk,
  input j,
  input k,
  input areset,
  output out
);
  // JK flip-flop with asynchronous active-high reset to 0
  reg state;

  always @(posedge clk or posedge areset) begin
    if (areset)
      state <= 1'b0;
    else
      state <= state ? ~k : j;
  end

  assign out = state;
endmodule
[DONE]
