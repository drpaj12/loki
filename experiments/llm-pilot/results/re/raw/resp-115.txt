[BEGIN]
module TopModule(
  input clk,
  input reset,
  input ena,
  output reg pm,
  output reg [7:0] hh,
  output reg [7:0] mm,
  output reg [7:0] ss
);
  // 12-hour BCD clock. Reset sets 12:00:00 AM.
  wire ss_wrap = (ss == 8'h59);
  wire mm_wrap = (mm == 8'h59);

  always @(posedge clk) begin
    if (reset) begin
      pm <= 1'b0;
      hh <= 8'h12;
      mm <= 8'h00;
      ss <= 8'h00;
    end else if (ena) begin
      // seconds
      if (ss_wrap) ss <= 8'h00;
      else if (ss[3:0] == 4'd9) ss <= {ss[7:4] + 4'd1, 4'd0};
      else ss <= ss + 8'd1;

      // minutes
      if (ss_wrap) begin
        if (mm_wrap) mm <= 8'h00;
        else if (mm[3:0] == 4'd9) mm <= {mm[7:4] + 4'd1, 4'd0};
        else mm <= mm + 8'd1;
      end

      // hours
      if (ss_wrap && mm_wrap) begin
        if (hh == 8'h12) hh <= 8'h01;
        else if (hh[3:0] == 4'd9) hh <= {hh[7:4] + 4'd1, 4'd0};
        else hh <= hh + 8'd1;
        if (hh == 8'h11) pm <= ~pm;
      end
    end
  end
endmodule
[DONE]
