module TopModule(
  input [2:0] in,
  output [1:0] out
);
  // This module counts the number of 1's in the 3-bit input vector
  // and outputs the count as a 2-bit binary number (0-3)
  // This is also known as a population count or hamming weight
  
  // Using clean, readable Verilog with direct assignments
  assign out[0] = in[0] ^ in[1] ^ in[2]; // LSB of count is 1 if odd number of 1's
